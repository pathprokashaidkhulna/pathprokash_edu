    function toBanglaNum(num) {
        if (!num) return '';
        const banglaDigits = {'0':'০','1':'১','2':'২','3':'৩','4':'৪','5':'৫','6':'৬','7':'৭','8':'৮','9':'৯'};
        return num.toString().replace(/\d/g, d => banglaDigits[d]);
    }

    function generateAutoRoll(selectedClass) {
        if (!selectedClass) {

            return;
        }
        let allStudents = JSON.parse(localStorage.getItem('path_prokash_master_db') || '[]');
        let sameClassStudents = allStudents.filter(s => s.targetClass === selectedClass);
        
        let startRoll = 5001;
        if (selectedClass.includes('৬ষ্ঠ')) startRoll = 6001;

    function calculateAge(dobString) {
        if (!dobString) { document.getElementById('ageResult').innerText = ''; return ''; }
        let dob = new Date(dobString);
        let today = new Date();
        let years = today.getFullYear() - dob.getFullYear();
        let months = today.getMonth() - dob.getMonth();
        let days = today.getDate() - dob.getDate();

        if (days < 0) {
            months--;
            let prevMonth = new Date(today.getFullYear(), today.getMonth(), 0);
            days += prevMonth.getDate();
        }
        if (months < 0) { years--; months += 12; }

        let ageText = `${toBanglaNum(years)} বছর ${toBanglaNum(months)} মাস ${toBanglaNum(days)} দিন`;
        document.getElementById('ageResult').innerText = "বর্তমান বয়স: " + ageText;
        return ageText;
    }

            return;
        }

        let allStudents = JSON.parse(localStorage.getItem('path_prokash_master_db') || '[]');

        let studentRecord = {
            targetClass: targetClass,
            autoRoll: autoRoll,
            nameBn: nameBn,
            nameEn: document.getElementById('inNameEn').value.trim().toUpperCase(),

            document.getElementById('outNameEn').innerText = studentRecord.nameEn;

            document.getElementById('outFatherEn').innerText = studentRecord.fatherEn;
            document.getElementById('outMotherBn').innerText = studentRecord.motherBn;
            document.getElementById('outMotherEn').innerText = studentRecord.motherEn;

    function exportToExcel() {
        if (!isAuthenticated) return;
        let allStudents = JSON.parse(localStorage.getItem('path_prokash_master_db') || '[]');
        if (allStudents.length === 0) {
            alert("ব্যাকআপ নেওয়ার জন্য কোনো তথ্য নেই!");
            return;
        }

        let excelContent = "\uFEFFঅটো রোল\tভর্তির শ্রেণি\tনাম (বাংলা)\tName (English)\tপিতার নাম\tমাতার নাম\tঠিকানা\tজন্ম তারিখ\tমোবাইল\tবিদ্যালয়\n";
        allStudents.forEach(s => {
            let address = `${s.union || ''}, ${s.upajela || ''}, ${s.jela || ''}`;
            excelContent += `${s.autoRoll || ''}\t${s.targetClass || ''}\t${s.nameBn || ''}\t${s.nameEn || ''}\t${s.fatherBn || ''}\t${s.motherBn || ''}\t${address}\t${s.dob || ''}\t${s.mobile || ''}\t${s.school || ''}\n`;
        });

        let blob = new Blob([excelContent], { type: 'application/vnd.ms-excel;charset=utf-8;' });
        let url = URL.createObjectURL(blob);
        let a = document.createElement('a');
        a.href = url;
        a.download = 'Path_Prokash_Master_Database.xls';
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
    }
